async function startVoice() { try { micBtn.classList.add('active'); micBtn.title = 'Stop voice'; micBtn.innerHTML = '⏹'; setStatus('connecting', 'Connecting...'); modeLabel.textContent = 'Voice mode'; sessionId = 'session_' + Date.now().toString(36); ws = new WebSocket(`${CONFIG.WS_BASE}/ws/voice/${sessionId}`); ws.onopen = async () => { setStatus('connected', 'Listening...'); ws.send(JSON.stringify({ type: 'start_session' })); startHeartbeat(); ac = new (window.AudioContext || window.webkitAudioContext)(); micStream = await navigator.mediaDevices.getUserMedia({ audio: { channelCount: 1, echoCancellation: true, noiseSuppression: true, autoGainControl: false } }); srcNode = ac.createMediaStreamSource(micStream); proc = ac.createScriptProcessor(4096, 1, 1); srcNode.connect(proc); proc.connect(ac.destination); proc.onaudioprocess = (e) => { const input = e.inputBuffer.getChannelData(0); let rms = 0; for (let i = 0; i < input.length; i++) rms += input[i] * input[i]; rms = Math.sqrt(rms / input.length); if (rms < 0.01) return; const pcm16 = new Int16Array(input.length); for (let i = 0; i < input.length; i++) { const s = Math.max(-1, Math.min(1, input[i])); pcm16[i] = s < 0 ? s * 0x8000 : s * 0x7fff; } const bytes = new Uint8Array(pcm16.buffer); let binary = '', chunkSize = 0x4000; for (let i = 0; i < bytes.length; i += chunkSize) binary += String.fromCharCode.apply(null, bytes.subarray(i, i + chunkSize)); if (ws && ws.readyState === WebSocket.OPEN) ws.send(JSON.stringify({ type: 'audio_data', data: btoa(binary) })); }; recording = true; }; ws.onmessage = (ev) => { const data = JSON.parse(ev.data); if (data.type === 'transcript_delta') addMsg('assistant', data.delta, true); if (data.type === 'transcript') addMsg('assistant', '🔊 ' + data.text); if (data.type === 'audio_response') playPCM16Hex(data.data); if (data.type === 'error') addMsg('sys', 'Voice error: ' + data.message); }; } catch (err) { addMsg('sys', 'Mic/WebSocket error: ' + err.message); stopVoice(); } }
